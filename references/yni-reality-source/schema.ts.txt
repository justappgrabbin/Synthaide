// ═══════════════════════════════════════════════════════════════════
// BIRTHDAY CALIBRATION SYSTEM - 4 Tier Accuracy Model
// Core rule: A birthday is Date + Time + Location + Timezone
// ═══════════════════════════════════════════════════════════════════

export const accuracyLevelEnum = z.enum([
  'date_only',           // Tier 0: Date only, assume 12:00 UTC
  'datetime_no_place',   // Tier 1: Date + time, no location
  'date_place_no_time',  // Tier 2: Date + place, no time (assume 12:00 local)
  'exact'                // Tier 3: Full calibration - Date + Time + Location + Timezone
]);

export const birthInputSchema = z.object({
  date: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, "Date must be YYYY-MM-DD format"),
  time: z.string().regex(/^\d{2}:\d{2}$/, "Time must be HH:MM format").optional(),
  city: z.string().optional(),
  region: z.string().optional(),
  country: z.string().optional(),
  latitude: z.number().min(-90).max(90).optional(),
  longitude: z.number().min(-180).max(180).optional(),
  timezone: z.string().optional(),
  utcOffsetHours: z.number().min(-12).max(14).optional(),
  dstInEffect: z.boolean().optional(),
  accuracyLevel: accuracyLevelEnum,
});

export const normalizedBirthDataSchema = z.object({
  unixTimestamp: z.number(),
  julianDay: z.number(),
  ageYears: z.number(),
  birthdaySeed: z.number().min(0).max(1),
  dayOfYear: z.number().min(1).max(366),
  dayMod64: z.number().min(0).max(63),
  dayMod384: z.number().min(0).max(383),
});

export const birthdayCalibrationSchema = z.object({
  id: z.string(),
  userId: z.string().optional(),
  birthInput: birthInputSchema,
  normalized: normalizedBirthDataSchema,
  calibrationTier: z.number().min(0).max(3),
  tierDescription: z.string(),
  warnings: z.array(z.string()).default([]),
  createdAt: z.number(),
  updatedAt: z.number(),
});

export const insertBirthdayCalibrationSchema = birthdayCalibrationSchema.omit({
  id: true,
  createdAt: true,
  updatedAt: true,
});

export type AccuracyLevel = z.infer<typeof accuracyLevelEnum>;

export type BirthdayCalibration = z.infer<typeof birthdayCalibrationSchema>;
export type InsertBirthdayCalibration = z.infer<typeof insertBirthdayCalibrationSchema>;

// ═══════════════════════════════════════════════════════════════════
// DMS (DEGREE-MINUTE-SECOND) PRECISION SYSTEM
// For exact gate/line calculations - NO decimals allowed
// Gate = 5° 37' 30" (20250 seconds), Line = 0° 56' 15" (3375 seconds)
// ═══════════════════════════════════════════════════════════════════

export const dmsSchema = z.object({
  deg: z.number().int().min(0).max(359),
  min: z.number().int().min(0).max(59),
  sec: z.number().min(0).max(60)
});

export const trinityFieldSchema = z.object({
  reference: z.enum(['tropical', 'sidereal_fagan_bradley', 'draconic_true_node']),
  orientation: z.enum(['gravity', 'electromagnetic', 'ambipolar_plasma']),
  positions_dms: z.record(z.string(), dmsSchema),
  gates: z.record(z.string(), z.number().min(1).max(64)),
  lines: z.record(z.string(), z.number().min(1).max(6)),
  colors: z.record(z.string(), z.number().min(1).max(6)),
  tones: z.record(z.string(), z.number().min(1).max(6)),
  bases: z.record(z.string(), z.number().min(1).max(5))
});

export const trinityCalibrationSchema = z.object({
  id: z.string(),
  userId: z.string().optional(),
  fields: z.object({
    body: trinityFieldSchema,
    mind: trinityFieldSchema,
    heart: trinityFieldSchema
  }),
  calibrationTimestamp: z.number(),
  accuracyLevel: accuracyLevelEnum,
  birthInput: birthInputSchema.optional()
});

export const insertTrinityCalibrationSchema = trinityCalibrationSchema.omit({
  id: true
});
