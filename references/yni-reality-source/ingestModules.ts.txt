// Helper to sanitize and validate filename
function sanitizeFilename(filename: string): string | null {
  // Reject filenames with path traversal attempts
  if (filename.includes('..') || filename.includes('/') || filename.includes('\\')) {
    console.warn(`Rejected potentially malicious filename: ${filename}`);
    return null;
  }
  
  // Only allow zip files
  if (!filename.toLowerCase().endsWith('.zip')) {
    return null;
  }
  
  return filename;
}

// Helper to clean module name from filename
function cleanModuleName(filename: string): string {
  // Remove file extension
  let name = filename.replace(/\.zip$/i, '');
  // Remove timestamp pattern _xxxxxxxxxx
  name = name.replace(/_\d{13}$/, '');
  // Remove (Copy) (1) (2) etc
  name = name.replace(/\s*\([^)]*\)$/g, '');
  return name;
}

// Helper to get metadata for a module
function getModuleMetadata(cleanName: string): ModuleMetadata | null {
  // Try exact match first
  if (moduleMetadataMap[cleanName]) {
    return moduleMetadataMap[cleanName];
  }
  
  // Try case-insensitive match
  const lowerName = cleanName.toLowerCase();
  for (const [key, value] of Object.entries(moduleMetadataMap)) {
    if (key.toLowerCase() === lowerName) {
      return value;
    }
  }
  
  // Generate default metadata
  return {
    id: cleanName.toLowerCase().replace(/[^a-z0-9]/g, ''),
    name: cleanName,
    version: "1.0.0",
    description: `Module: ${cleanName}`,
    category: "utility",
    tags: ["module"],
  };
}

export function ingestModulesFromDirectory(modulesPath: string): InsertModuleCatalog[] {
  const catalogModules: InsertModuleCatalog[] = [];
  const processedNames = new Set<string>();

  try {
    const files = readdirSync(modulesPath);

    for (const file of files) {
      if (!file.endsWith('.zip')) continue;

      // Sanitize filename to prevent path traversal
      const sanitizedFile = sanitizeFilename(file);
      if (!sanitizedFile) {
        console.warn(`Skipping invalid or potentially malicious file: ${file}`);
        continue;
      }

      const cleanName = cleanModuleName(sanitizedFile);
      
      // Skip if we've already processed this module (deduplication)
      if (processedNames.has(cleanName.toLowerCase())) {
        continue;
      }
      
      processedNames.add(cleanName.toLowerCase());

      const filePath = join(modulesPath, sanitizedFile);
      const stats = statSync(filePath);
      const metadata = getModuleMetadata(cleanName);

      if (!metadata) continue;

      // Only allow files within the modules directory (security check)
      const normalizedPath = join('/modules', sanitizedFile).replace(/\\/g, '/');

      catalogModules.push({
        id: metadata.id,
        name: metadata.name,
        version: metadata.version,
        description: metadata.description,
        category: metadata.category,
        author: metadata.author,
        tags: metadata.tags,
        filePath: normalizedPath,
        fileSize: stats.size,
        gates: metadata.gates,
        requiredFields: metadata.requiredFields,
        installStatus: "available",
      });
    }
  } catch (error) {
    console.error('Error ingesting modules:', error);
  }

  return catalogModules;
}
