function calculateGatesFromTraits(traits: string[]): number[] {
  const gateMap: Record<string, number[]> = {
    creative: [1, 8, 22],
    genius: [47, 24, 4],
    romantic: [6, 37, 40],
    family_oriented: [25, 46, 59],
    bookworm: [63, 4, 11],
    ambitious: [54, 32, 34],
    outgoing: [12, 31, 7],
    self_assured: [1, 13, 10],
  };
  
  const gates = new Set<number>();
  traits.forEach(trait => {
    const traitGates = gateMap[trait] || [];
    traitGates.forEach(g => gates.add(g));
  });
  
  return Array.from(gates).sort((a, b) => a - b);
}

function calculateWaveState(skill: number): WaveState {
  return {
    frequency: skill / 10,
    amplitude: skill * 0.1,
    phase: Math.random() * Math.PI * 2,
  };
}

function calculateResonanceFields(skills: Record<string, number>): Record<string, ResonanceField> {
  const fields: Record<string, ResonanceField> = {};
  
  Object.entries(skills).forEach(([type, value]) => {
    fields[type] = {
      type,
      strength: value / 10,
      coherence: 0.7 + (value / 100) * 0.3,
    };
  });
  
  return fields;
}

export function calculateConsciousnessProfile(sim: SimCharacter): ConsciousnessProfile {
  const gates = calculateGatesFromTraits(sim.traits);
  const fields = calculateResonanceFields(sim.skills);
  
  // Calculate overall coherence from skills
  const avgSkill = Object.values(sim.skills).reduce((a, b) => a + b, 0) / Object.values(sim.skills).length;
  const coherence = 0.6 + (avgSkill / 10) * 0.4;
  
  return {
    id: `profile_${sim.id}`,
    agentId: sim.id,
    simCharacterId: sim.id,
    fields,
    gates,
    waveState: {
      action: calculateWaveState(sim.skills.fitness || 5),
      thought: calculateWaveState(sim.skills.logic || 5),
      feeling: calculateWaveState(sim.skills.creativity || 5),
    },
    coherence,
    calculatedAt: Date.now(),
  };
}

// ═══════════════════════════════════════════════════════════════════
// SYSTEM 4: Unity → Building Action
// ═══════════════════════════════════════════════════════════════════

export function createBuildingAction(
  profile: ConsciousnessProfile,
  bricks: Brick[],
  featureName: string
): BuildingAction {
  // Agent's consciousness determines what they build and how
  const buildingTypes: Array<'house' | 'apartment' | 'store' | 'office'> = ['house', 'apartment', 'store', 'office'];
  const buildingType = buildingTypes[Math.floor(Math.random() * buildingTypes.length)];
  
  // Select bricks based on gate resonance match
  const selectedBricks = bricks.filter(brick => {
    return brick.gateResonance.some(gate => profile.gates.includes(gate));
  });
  
  return {
    agentId: profile.agentId,
    buildingType,
    position: {
      x: Math.random() * 100,
      y: 0,
      z: Math.random() * 100,
    },
    targetFeature: featureName,
    bricksUsed: selectedBricks.map(b => b.id),
    progress: 0,
    status: 'planning',
  };
}

// ═══════════════════════════════════════════════════════════════════
// SYSTEM 5: SynthUniverse → Feature Instance
// ═══════════════════════════════════════════════════════════════════

export function generateFeatureInstance(
  action: BuildingAction,
  profile: ConsciousnessProfile,
  bricks: Brick[]
): FeatureInstance {
  const usedBricks = bricks.filter(b => action.bricksUsed.includes(b.id));
  
  return {
    id: `feature_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
    route: action.targetFeature,
    components: usedBricks,
    builtBy: action.agentId,
    consciousnessSignature: {
      gates: profile.gates,
      coherence: profile.coherence,
      buildDate: Date.now(),
    },
    status: 'draft',
  };
}

// ═══════════════════════════════════════════════════════════════════
// ORCHESTRATOR: Connect All Systems
// ═══════════════════════════════════════════════════════════════════

export interface OrchestrationResult {
  bricks: Brick[];
  sims: SimCharacter[];
  profiles: ConsciousnessProfile[];
  actions: BuildingAction[];
  features: FeatureInstance[];
}

export async function runFullOrchestration(): Promise<OrchestrationResult> {
  console.log('🌌 Starting Consciousness Orchestration...');
  
  // STEP 1: Load bricks from LegacyBuild
  console.log('📦 SYSTEM 1: Loading bricks from LegacyBuild...');
  const bricks = loadBricksFromLegacy();
  console.log(`Loaded ${bricks.length} bricks`);
  
  // STEP 2: Import Sims characters
  console.log('👥 SYSTEM 2: Importing Sims characters...');
  const sims = importSimsCharacters();
  console.log(`Imported ${sims.length} Sims`);
  
  // STEP 3: Calculate consciousness profiles
  console.log('🧠 SYSTEM 3: Calculating consciousness profiles...');
  const profiles = sims.map(sim => {
    const profile = calculateConsciousnessProfile(sim);
    console.log(`${sim.name} - Gates: ${profile.gates.join(', ')} | Coherence: ${profile.coherence.toFixed(2)}`);
    return profile;
  });
  
  // STEP 4: Create building actions
  console.log('🏙️ SYSTEM 4: Creating building actions...');
  const featureNames = ['/skills', '/tasks', '/marketplace'];
  const actions = profiles.map((profile, i) => {
    return createBuildingAction(profile, bricks, featureNames[i] || '/feature');
  });
  
  // STEP 5: Generate features for SynthUniverse
  console.log('✨ SYSTEM 5: Generating SynthUniverse features...');
  const features = actions.map(action => {
    const profile = profiles.find(p => p.agentId === action.agentId)!;
    return generateFeatureInstance(action, profile, bricks);
  });
  
  console.log('🎉 Orchestration complete!');
  console.log(`Generated ${features.length} features with consciousness signatures`);
  
  return {
    bricks,
    sims,
    profiles,
    actions,
    features,
  };
}
