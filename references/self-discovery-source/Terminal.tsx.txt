export function Terminal() {
  const { setPlayerName } = useGame();
  const [visibleLines, setVisibleLines] = useState<TerminalLine[]>([]);
  const [pendingLines, setPendingLines] = useState<TerminalLine[]>(TERMINAL_STORY);
  const [showNameInput, setShowNameInput] = useState(false);
  const [name, setName] = useState('');
  const [isComplete, setIsComplete] = useState(false);
  const [isInTriggerPhase, setIsInTriggerPhase] = useState(false);
  const timeoutsRef = useRef<NodeJS.Timeout[]>([]);

  const clearTimeouts = useCallback(() => {
    timeoutsRef.current.forEach(t => clearTimeout(t));
    timeoutsRef.current = [];
  }, []);

  useEffect(() => {
    return () => clearTimeouts();
  }, [clearTimeouts]);

  useEffect(() => {
    if (pendingLines.length === 0) {
      if (isInTriggerPhase) {
        setTimeout(() => setShowNameInput(true), 500);
      }
      return;
    }

    clearTimeouts();

    pendingLines.forEach((line, index) => {
      const timeout = setTimeout(() => {
        setVisibleLines(prev => {
          if (prev.some(l => l.id === line.id)) return prev;
          return [...prev, line];
        });
      }, line.delay || 0);
      timeoutsRef.current.push(timeout);
    });

    const maxDelay = Math.max(...pendingLines.map(l => l.delay || 0));
    const completionTimeout = setTimeout(() => {
      setPendingLines([]);
    }, maxDelay + 100);
    timeoutsRef.current.push(completionTimeout);

  }, [pendingLines, clearTimeouts, isInTriggerPhase]);

  const handleKeywordClick = useCallback((action: string) => {
    clearTimeouts();
    
    if (action === 'trigger') {
      setIsInTriggerPhase(true);
      setPendingLines(STORY_BRANCHES.trigger);
    } else if (STORY_BRANCHES[action]) {
      setPendingLines(STORY_BRANCHES[action]);
    }
  }, [clearTimeouts]);

  const handleNameSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (name.trim()) {
      setShowNameInput(false);
      setIsComplete(true);
      setPlayerName(name.trim());
    }
  };

  const renderLine = (line: TerminalLine) => {
    if (line.keywords && line.keywords.length > 0) {
      const parts = line.text.split(/(\[.*?\])/g);
      return (
        <span>
          {parts.map((part, i) => {
            const match = part.match(/\[(.*?)\]/);
            if (match) {
              const keyword = line.keywords?.find(k => match[1].includes(k.word));
              if (keyword) {
                return (
                  <button
                    key={`${line.id}-btn-${i}`}
                    onClick={() => handleKeywordClick(keyword.action)}
                    className="keyword-highlight mx-2 text-cyan font-bold uppercase tracking-wider hover:text-glow-cyan transition-all"
                    data-testid={`keyword-${keyword.word.toLowerCase()}`}
                  >
                    [{keyword.word}]
                  </button>
                );
              }
            }
            return <span key={`${line.id}-txt-${i}`}>{part}</span>;
          })}
        </span>
      );
    }
    return line.text;
  };

  return (
    <div className="min-h-screen bg-background flex items-center justify-center p-4 md:p-8">
      <div className="scanline-overlay" />
      
      <div className="w-full max-w-4xl">
        <div 
          className="bg-card/80 backdrop-blur-sm border border-cyan/20 rounded-md p-6 md:p-8 shadow-lg"
          data-testid="terminal-container"
        >
          <div className="mb-6 flex items-center gap-2">
            <div className="w-3 h-3 rounded-full bg-glitch-red" />
            <div className="w-3 h-3 rounded-full bg-yellow-500" />
            <div className="w-3 h-3 rounded-full bg-bio-green" />
            <span className="ml-4 text-muted-foreground text-sm uppercase tracking-widest">
              synthworld://consciousness.interface
            </span>
          </div>

          <div className="font-mono text-sm md:text-base leading-relaxed space-y-1 min-h-[400px] max-h-[600px] overflow-y-auto">
            {visibleLines.map((line, index) => (
              <div 
                key={line.id} 
                className={`fade-in ${
                  line.type === 'system' 
                    ? 'text-muted-foreground' 
                    : line.type === 'keyword'
                    ? 'text-foreground py-2'
                    : 'text-foreground'
                } ${line.text.includes('WARNING') || line.text.includes('DESTABILIZING') ? 'text-glitch-red' : ''}
                ${line.text.includes('Y̷O̴U̷') || line.text.includes('A̸R̵E̶') || line.text.includes('L̴O̷S̵T̸') ? 'text-magenta text-xl' : ''}`}
                style={{ animationDelay: `${index * 30}ms` }}
              >
                {renderLine(line)}
              </div>
            ))}
            
            {showNameInput && (
              <form onSubmit={handleNameSubmit} className="mt-8 fade-in">
                <div className="flex flex-col gap-4">
                  <label className="text-cyan uppercase tracking-widest text-sm">
                    IDENTITY FRAGMENT REQUIRED
                  </label>
                  <p className="text-muted-foreground text-sm">
                    Before we can begin recovery, we need something to call you.
                  </p>
                  <div className="flex gap-4">
                    <Input
                      type="text"
                      value={name}
                      onChange={(e) => setName(e.target.value)}
                      placeholder="Enter your name..."
                      className="bg-background/50 border-cyan/30 text-foreground placeholder:text-muted-foreground focus:border-cyan"
                      autoFocus
                      data-testid="input-player-name"
                    />
                    <Button 
                      type="submit" 
                      className="bg-cyan/20 border border-cyan text-cyan hover:bg-cyan hover:text-background"
                      data-testid="button-start-journey"
                    >
                      BEGIN
                    </Button>
                  </div>
                </div>
              </form>
            )}

            {isComplete && (
              <div className="mt-8 text-center fade-in">
                <p className="text-cyan text-xl text-glow-cyan">
                  WELCOME, {name.toUpperCase()}
                </p>
                <p className="text-muted-foreground mt-2">
                  Initializing consciousness matrix...
                </p>
              </div>
            )}

            {!showNameInput && !isComplete && pendingLines.length === 0 && !isInTriggerPhase && (
              <span className="terminal-cursor" />
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
