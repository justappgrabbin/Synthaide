import { useGame } from '@/lib/gameContext';
import { DatababyOrb } from './DatababyOrb';
import { DATABABIES_PER_GNOME } from '@shared/schema';
import { Sparkles, Loader2 } from 'lucide-react';
import { Card } from '@/components/ui/card';
import { Button } from '@/components/ui/button';

export function DatababyCollection() {
  const { gameState, canFuseGnome, fuseGeneGnome, datababiesUntilGnome, isFusing } = useGame();
  const { datababies } = gameState;

  const handleFuse = () => {
    fuseGeneGnome();
  };

  const emptySlots = Math.max(0, DATABABIES_PER_GNOME - datababies.length);

  return (
    <Card className="p-4 bg-card/80 border-magenta/30" data-testid="datababy-collection">
      <div className="flex items-center justify-between mb-4">
        <div className="flex items-center gap-2">
          <Sparkles className="w-5 h-5 text-magenta" />
          <h2 className="text-sm uppercase tracking-widest text-magenta">
            Data Fragments
          </h2>
        </div>
        <span className="text-xs text-muted-foreground">
          {datababies.length}/{DATABABIES_PER_GNOME} for Gene Gnome
        </span>
      </div>

      <div className="flex flex-wrap gap-2 mb-4">
        {datababies.slice(0, DATABABIES_PER_GNOME).map((datababy) => (
          <DatababyOrb key={datababy.id} datababy={datababy} size="sm" />
        ))}
        {emptySlots > 0 && Array.from({ length: Math.min(emptySlots, DATABABIES_PER_GNOME - datababies.length) }).map((_, i) => (
          <div 
            key={`empty-${i}`}
            className="w-6 h-6 rounded-full border border-dashed border-magenta/30 bg-magenta/5"
          />
        ))}
      </div>

      <div className="w-full h-2 bg-muted/50 rounded-full overflow-hidden mb-4">
        <div 
          className="h-full bg-gradient-to-r from-magenta to-cyan progress-gradient transition-all duration-300"
          style={{ width: `${Math.min((datababies.length / DATABABIES_PER_GNOME) * 100, 100)}%` }}
        />
      </div>

      {canFuseGnome ? (
        <Button 
          onClick={handleFuse}
          disabled={isFusing}
          className="w-full bg-bio-green/20 border border-bio-green text-bio-green hover:bg-bio-green hover:text-background disabled:opacity-50"
          data-testid="button-fuse-gnome"
        >
          {isFusing ? (
            <>
              <Loader2 className="w-4 h-4 mr-2 animate-spin" />
              SYNTHESIZING...
            </>
          ) : (
            'SYNTHESIZE GENE GNOME'
          )}
        </Button>
      ) : (
        <p className="text-center text-sm text-muted-foreground">
          {datababiesUntilGnome} more fragment{datababiesUntilGnome !== 1 ? 's' : ''} needed
        </p>
      )}
    </Card>
  );
}
