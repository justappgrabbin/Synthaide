import { createContext, useContext, useState, useCallback, useEffect, type ReactNode } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { apiRequest } from './queryClient';
import type { 
  GameState, 
  Datababy, 
  GeneGnome, 
  DiagnosticLog,
} from '@shared/schema';
import { DATABABIES_PER_GNOME } from '@shared/schema';

const PLAYER_ID_KEY = 'synthworld_player_id';

const INITIAL_GAME_STATE: GameState = {
  playerId: '',
  playerName: '',
  level: 1,
  coherence: 100,
  maxCoherence: 100,
  datababies: [],
  geneGnomes: [],
  fieldFriends: [],
  hasAgent: false,
  terminalCompleted: false,
  currentPhase: 'terminal',
  diagnosticLogs: [],
  totalDatababiesCollected: 0,
  glitchActive: false,
};

interface GameContextType {
  gameState: GameState;
  isLoading: boolean;
  error: Error | null;
  setPlayerName: (name: string) => void;
  completeTerminal: () => void;
  collectDatababy: (category: Datababy['category']) => void;
  fuseGeneGnome: () => void;
  updateCoherence: (change: number) => void;
  addDiagnosticLog: (type: DiagnosticLog['type'], message: string) => void;
  triggerGlitch: () => void;
  resetGame: () => void;
  canFuseGnome: boolean;
  datababiesUntilGnome: number;
  isCollecting: boolean;
  isFusing: boolean;
}

const GameContext = createContext<GameContextType | null>(null);

export function GameProvider({ children }: { children: ReactNode }) {
  const queryClient = useQueryClient();
  const [localState, setLocalState] = useState<GameState>(INITIAL_GAME_STATE);
  const [playerId, setPlayerId] = useState<string | null>(() => {
    if (typeof window !== 'undefined') {
      return localStorage.getItem(PLAYER_ID_KEY);
    }
    return null;
  });

  const { data: serverGameState, isLoading, error } = useQuery<GameState>({
    queryKey: ['/api/game', playerId],
    enabled: !!playerId,
  });

  const isInGamePhase = localState.currentPhase === 'game' || localState.currentPhase === 'awakening';
  
  const gameState: GameState = {
    ...INITIAL_GAME_STATE,
    ...(serverGameState || {}),
    currentPhase: localState.currentPhase,
    terminalCompleted: localState.terminalCompleted || serverGameState?.terminalCompleted || false,
    playerName: localState.playerName || serverGameState?.playerName || '',
    coherence: localState.coherence,
    glitchActive: localState.glitchActive,
    datababies: localState.datababies.length > 0 ? localState.datababies : (serverGameState?.datababies || []),
    geneGnomes: localState.geneGnomes.length > 0 ? localState.geneGnomes : (serverGameState?.geneGnomes || []),
    fieldFriends: localState.fieldFriends.length > 0 ? localState.fieldFriends : (serverGameState?.fieldFriends || []),
    diagnosticLogs: localState.diagnosticLogs.length > 0 ? localState.diagnosticLogs : (serverGameState?.diagnosticLogs || []),
    level: localState.level > 1 ? localState.level : (serverGameState?.level || 1),
    totalDatababiesCollected: Math.max(localState.totalDatababiesCollected, serverGameState?.totalDatababiesCollected || 0),
  };

  useEffect(() => {
    if (serverGameState && !isInGamePhase) {
      setLocalState(prev => ({
        ...prev,
        ...serverGameState,
        currentPhase: prev.currentPhase,
      }));
    }
  }, [serverGameState, isInGamePhase]);

  const createGameMutation = useMutation({
    mutationFn: async (playerName: string) => {
      const response = await apiRequest('POST', '/api/game', { playerName });
      return response.json() as Promise<GameState>;
    },
    onSuccess: (data) => {
      setPlayerId(data.playerId);
      localStorage.setItem(PLAYER_ID_KEY, data.playerId);
      setLocalState(prev => ({
        ...data,
        currentPhase: 'awakening',
        terminalCompleted: true,
        coherence: prev.coherence,
      }));
    },
  });

  const updateGameMutation = useMutation({
    mutationFn: async (updates: Partial<GameState>) => {
      if (!playerId) throw new Error('No player ID');
      const response = await apiRequest('PATCH', `/api/game/${playerId}`, updates);
      return response.json() as Promise<GameState>;
    },
    onSuccess: (data) => {
      setLocalState(prev => ({
        ...prev,
        ...data,
        currentPhase: prev.currentPhase,
        terminalCompleted: prev.terminalCompleted,
        coherence: prev.coherence,
        glitchActive: prev.glitchActive,
      }));
    },
  });

  const collectDatababyMutation = useMutation({
    mutationFn: async (category: Datababy['category']) => {
      if (!playerId) throw new Error('No player ID');
      const response = await apiRequest('POST', `/api/game/${playerId}/datababy`, { category });
      return response.json() as Promise<{ gameState: GameState; datababy: Datababy }>;
    },
    onSuccess: (data) => {
      setLocalState(prev => ({
        ...prev,
        datababies: data.gameState.datababies,
        totalDatababiesCollected: data.gameState.totalDatababiesCollected,
        diagnosticLogs: data.gameState.diagnosticLogs,
      }));
    },
  });

  const fuseGnomeMutation = useMutation({
    mutationFn: async () => {
      if (!playerId) throw new Error('No player ID');
      const response = await apiRequest('POST', `/api/game/${playerId}/fuse`, {});
      return response.json() as Promise<{ gameState: GameState; geneGnome: GeneGnome }>;
    },
    onSuccess: (data) => {
      setLocalState(prev => ({
        ...prev,
        datababies: data.gameState.datababies,
        geneGnomes: data.gameState.geneGnomes,
        level: data.gameState.level,
        maxCoherence: data.gameState.maxCoherence,
        diagnosticLogs: data.gameState.diagnosticLogs,
      }));
    },
  });

  const updateCoherenceMutation = useMutation({
    mutationFn: async (change: number) => {
      if (!playerId) throw new Error('No player ID');
      const response = await apiRequest('PATCH', `/api/game/${playerId}/coherence`, { change });
      return response.json() as Promise<GameState>;
    },
    onSuccess: (data) => {
      setLocalState(prev => ({
        ...prev,
        coherence: data.coherence,
        glitchActive: data.glitchActive,
        diagnosticLogs: data.diagnosticLogs,
      }));
    },
  });

  const addLogMutation = useMutation({
    mutationFn: async ({ type, message }: { type: DiagnosticLog['type']; message: string }) => {
      if (!playerId) throw new Error('No player ID');
      const response = await apiRequest('POST', `/api/game/${playerId}/log`, { type, message });
      return response.json() as Promise<GameState>;
    },
    onSuccess: (data) => {
      setLocalState(prev => ({
        ...prev,
        diagnosticLogs: data.diagnosticLogs,
      }));
    },
  });

  const setPlayerName = useCallback((name: string) => {
    createGameMutation.mutate(name);
    
    setTimeout(() => {
      setLocalState(prev => ({
        ...prev,
        terminalCompleted: true,
        currentPhase: 'game',
        coherence: 50,
      }));
    }, 3000);
  }, [createGameMutation]);

  const completeTerminal = useCallback(() => {
    setLocalState(prev => ({
      ...prev,
      terminalCompleted: true,
      currentPhase: 'awakening',
    }));
    
    setTimeout(() => {
      setLocalState(prev => ({
        ...prev,
        terminalCompleted: true,
        currentPhase: 'game',
        coherence: 50,
      }));
    }, 3000);
  }, []);

  const collectDatababy = useCallback((category: Datababy['category']) => {
    if (playerId) {
      collectDatababyMutation.mutate(category);
    } else {
      const fragments: Record<Datababy['category'], string[]> = {
        identity: ['I am', 'uniquely', 'myself', 'authentic', 'growing'],
        purpose: ['I create', 'meaning', 'through', 'my actions'],
        connection: ['I connect', 'with others', 'deeply', 'genuinely'],
        strength: ['I possess', 'courage', 'resilience', 'determination'],
        dream: ['I envision', 'a world', 'where', 'possibilities'],
      };
      const fragment = fragments[category][Math.floor(Math.random() * fragments[category].length)];
      const datababy: Datababy = {
        id: Math.random().toString(36).substring(2, 15),
        fragment,
        category,
        collectedAt: Date.now(),
      };
      setLocalState(prev => ({
        ...prev,
        datababies: [...prev.datababies, datababy],
        totalDatababiesCollected: prev.totalDatababiesCollected + 1,
        coherence: Math.min(prev.maxCoherence, prev.coherence + 2),
      }));
    }
  }, [playerId, collectDatababyMutation]);

  const fuseGeneGnome = useCallback(() => {
    if (gameState.datababies.length < DATABABIES_PER_GNOME) return;
    
    if (playerId) {
      fuseGnomeMutation.mutate();
    } else {
      const datababiesToFuse = gameState.datababies.slice(0, DATABABIES_PER_GNOME);
      const sentence = datababiesToFuse.map(d => d.fragment).join(' ') + '.';
      const archetypes = ['Memory Keeper', 'Dream Weaver', 'Pattern Finder', 'Truth Speaker'];
      const archetype = archetypes[Math.floor(Math.random() * archetypes.length)];
      
      const geneGnome: GeneGnome = {
        id: Math.random().toString(36).substring(2, 15),
        sentence,
        archetype,
        datababyIds: datababiesToFuse.map(d => d.id),
        fusedAt: Date.now(),
      };
      
      setLocalState(prev => ({
        ...prev,
        datababies: prev.datababies.slice(DATABABIES_PER_GNOME),
        geneGnomes: [...prev.geneGnomes, geneGnome],
        level: Math.min(72, prev.level + 1),
      }));
    }
  }, [playerId, gameState.datababies, fuseGnomeMutation]);

  const updateCoherence = useCallback((change: number) => {
    if (playerId) {
      updateCoherenceMutation.mutate(change);
    } else {
      setLocalState(prev => {
        const newCoherence = Math.max(0, Math.min(prev.maxCoherence, prev.coherence + change));
        return {
          ...prev,
          coherence: newCoherence,
          glitchActive: newCoherence < 30,
        };
      });
    }
  }, [playerId, updateCoherenceMutation]);

  const addDiagnosticLog = useCallback((type: DiagnosticLog['type'], message: string) => {
    const log: DiagnosticLog = {
      id: Math.random().toString(36).substring(2, 15),
      timestamp: Date.now(),
      type,
      message,
    };
    setLocalState(prev => ({
      ...prev,
      diagnosticLogs: [...prev.diagnosticLogs.slice(-50), log],
    }));
  }, []);

  const triggerGlitch = useCallback(() => {
    setLocalState(prev => ({ ...prev, glitchActive: true }));
    addDiagnosticLog('glitch', 'R̷E̵S̴O̷N̵A̴N̸C̶E̷ ̷U̷N̶S̶T̶A̴B̷L̷E̴');
    
    setTimeout(() => {
      setLocalState(prev => ({ ...prev, glitchActive: false }));
    }, 2000);
  }, [addDiagnosticLog]);

  const resetGame = useCallback(() => {
    localStorage.removeItem(PLAYER_ID_KEY);
    setPlayerId(null);
    setLocalState(INITIAL_GAME_STATE);
    queryClient.clear();
  }, [queryClient]);

  const canFuseGnome = gameState.datababies.length >= DATABABIES_PER_GNOME;
  const datababiesUntilGnome = DATABABIES_PER_GNOME - (gameState.datababies.length % DATABABIES_PER_GNOME);

  return (
    <GameContext.Provider value={{
      gameState,
      isLoading,
      error: error as Error | null,
      setPlayerName,
      completeTerminal,
      collectDatababy,
      fuseGeneGnome,
      updateCoherence,
      addDiagnosticLog,
      triggerGlitch,
      resetGame,
      canFuseGnome,
      datababiesUntilGnome,
      isCollecting: collectDatababyMutation.isPending,
      isFusing: fuseGnomeMutation.isPending,
    }}>
      {children}
    </GameContext.Provider>
  );
}

export function useGame() {
  const context = useContext(GameContext);
  if (!context) {
    throw new Error('useGame must be used within a GameProvider');
  }
  return context;
}

export function useGenerateDatababy() {
  const { collectDatababy } = useGame();
  return (category: Datababy['category']) => {
    collectDatababy(category);
  };
}
