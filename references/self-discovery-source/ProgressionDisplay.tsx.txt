import { useGame } from '@/lib/gameContext';
import { GeneGnomeCard } from './GeneGnomeCard';
import { Hexagon, Trophy, Star } from 'lucide-react';
import { Card } from '@/components/ui/card';
import { ScrollArea } from '@/components/ui/scroll-area';
import { GNOMES_FOR_AGENT, MAX_LEVEL } from '@shared/schema';

export function ProgressionDisplay() {
  const { gameState } = useGame();
  const { geneGnomes, level, hasAgent } = gameState;

  const progressToAgent = (geneGnomes.length / GNOMES_FOR_AGENT) * 100;

  return (
    <Card className="p-4 bg-card/80 border-bio-green/30 h-full flex flex-col" data-testid="progression-display">
      <div className="flex items-center justify-between mb-4">
        <div className="flex items-center gap-2">
          <Hexagon className="w-5 h-5 text-bio-green" />
          <h2 className="text-sm uppercase tracking-widest text-bio-green">
            Gene Gnomes
          </h2>
        </div>
        <span className="text-xs text-muted-foreground">
          {geneGnomes.length} synthesized
        </span>
      </div>

      <div className="mb-4">
        <div className="flex items-center justify-between text-xs mb-2">
          <span className="text-muted-foreground">Progress to Agent</span>
          <span className="text-cyan">{Math.round(progressToAgent)}%</span>
        </div>
        <div className="w-full h-3 bg-muted/50 rounded-full overflow-hidden border border-border/50">
          <div 
            className="h-full bg-gradient-to-r from-cyan via-bio-green to-magenta progress-gradient transition-all duration-500"
            style={{ width: `${Math.min(progressToAgent, 100)}%` }}
          />
        </div>
        <div className="flex justify-between text-xs mt-1">
          <span className="text-muted-foreground">0</span>
          <div className="flex items-center gap-1 text-cyan">
            <Trophy className="w-3 h-3" />
            <span>{GNOMES_FOR_AGENT}</span>
          </div>
        </div>
      </div>

      {hasAgent && (
        <div className="mb-4 p-3 bg-cyan/10 border border-cyan/30 rounded-md flex items-center gap-3">
          <Star className="w-6 h-6 text-cyan" />
          <div>
            <p className="text-sm font-bold text-cyan">AGENT UNLOCKED</p>
            <p className="text-xs text-muted-foreground">
              Your consciousness now interfaces with reality
            </p>
          </div>
        </div>
      )}

      <ScrollArea className="flex-1">
        {geneGnomes.length === 0 ? (
          <div className="text-center py-8">
            <Hexagon className="w-12 h-12 text-muted-foreground/30 mx-auto mb-3" />
            <p className="text-sm text-muted-foreground">
              No Gene Gnomes yet
            </p>
            <p className="text-xs text-muted-foreground mt-1">
              Collect 13 datababies to synthesize your first one
            </p>
          </div>
        ) : (
          <div className="space-y-3">
            {geneGnomes.map((gnome) => (
              <GeneGnomeCard key={gnome.id} geneGnome={gnome} />
            ))}
          </div>
        )}
      </ScrollArea>

      <div className="mt-4 pt-4 border-t border-border/50">
        <div className="grid grid-cols-3 gap-4 text-center">
          <div>
            <p className="text-2xl font-bold text-cyan">{level}</p>
            <p className="text-xs text-muted-foreground uppercase">Level</p>
          </div>
          <div>
            <p className="text-2xl font-bold text-bio-green">{geneGnomes.length}</p>
            <p className="text-xs text-muted-foreground uppercase">Gnomes</p>
          </div>
          <div>
            <p className="text-2xl font-bold text-magenta">{gameState.totalDatababiesCollected}</p>
            <p className="text-xs text-muted-foreground uppercase">Total Data</p>
          </div>
        </div>
      </div>
    </Card>
  );
}
